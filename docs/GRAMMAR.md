# Грамматика

EBNF. `{ x }` — ноль или больше, `[ x ]` — необязательно, `|` — выбор.

```ebnf
программа        = { объявление } "береста" блок EOF ;

объявление       = константа | функция | уклад ;

уклад            = "уклад" имя "{" { поле } "}" ;
поле             = имя ":" тип ";" ;

константа        = "устой" имя ":" тип "=" выражение ";" ;

функция          = "дело" имя "(" [ параметры ] ")" "->" тип блок ;

параметры        = параметр { "," параметр } ;
параметр         = имя ":" тип_объявления ;

тип              = "число" | "дробь" | "строка" | "флаг" | "ничто" ;

блок             = "{" { оператор } "}" ;

оператор         = переменная
                 | присваивание
                 | печать
                 | ветвление
                 | цикл_условный
                 | цикл_числовой
                 | обход
                 | возврат
                 | прервать
                 | продолжить
                 | блок
                 | выражение ";"
                 ;

переменная       = "слово" имя ":" тип_объявления [ "=" выражение ] ";" ;

тип_объявления   = тип | "[" целое "]" тип | "указатель" тип_элемента ;
тип_элемента     = тип | имя ;

присваивание     = имя "=" выражение ";"
                 | имя "[" выражение "]" "=" выражение ";"
                 | имя "." имя "=" выражение ";"
                 | "по" унарное "=" выражение ";"
                 | имя ("+=" | "-=" | "*=" | "/=" | "%=") выражение ";" ;

печать           = "молви" выражение ";" ;

ветвление        = "ежели" выражение блок
                   [ "инако" ( блок | ветвление ) ] ;

цикл_условный    = "доколе" выражение блок ;

цикл_числовой    = "повтор" имя "от" выражение "до" выражение
                   [ "шаг" выражение ] блок ;

обход            = "для" "каждого" имя "из" имя блок ;

возврат          = "вернуть" [ выражение ] ";" ;

прервать         = "прервать" ";" ;
продолжить       = "продолжить" ";" ;

выражение        = дизъюнкция ;

дизъюнкция       = конъюнкция { "или" конъюнкция } ;
конъюнкция       = равенство { "и" равенство } ;
равенство        = сравнение { ( "==" | "!=" ) сравнение } ;
сравнение        = сумма { ( "<" | "<=" | ">" | ">=" ) сумма } ;
сумма            = произведение { ( "+" | "-" ) произведение } ;
произведение     = унарное { ( "*" | "/" | "%" ) унарное } ;

унарное          = ( "-" | "!" | "не" ) унарное
                 | "адрес" унарное
                 | "по" унарное
                 | первичное ;

первичное        = целое
                 | дробное
                 | строка
                 | "правда"
                 | "кривда"
                 | имя [ "(" [ аргументы ] ")" ]
                 | имя "[" выражение "]"
                 | имя "." имя
                 | "(" выражение ")"
                 ;

аргументы        = выражение { "," выражение } ;

имя              = идентификатор ;
```

## Замечания к разбору

- `инако ежели` разбирается как вложенное `ежели`, а не как отдельная
  конструкция. Поэтому цепочка `инако ежели` может быть любой длины.
- Присваивание — оператор, а не выражение: `х = у = 1;` не разбирается.
- После `вернуть` пустое значение распознаётся по следующему `;`.
- `{` в позиции выражения — ошибка с подсказкой.
